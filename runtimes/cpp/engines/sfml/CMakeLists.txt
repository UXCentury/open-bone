cmake_minimum_required(VERSION 3.18)
project(openbone_sfml LANGUAGES C CXX)

# SFML 2.6 sample renderer. Needs SFML 2.6 installed (find_package) or -DOPENBONE_FETCH_SFML=ON to build it from
# source (on Linux that needs the X11, Xrandr, Xcursor, udev, OpenGL and FreeType development packages).
#
# Sound uses sfml-audio when it is available (OPENBONE_SFML_AUDIO, on by default). SFML 2's audio module links
# OpenAL Soft (LGPL-2.1, as a shared library) plus libogg / libvorbis / libFLAC (BSD); building it from source on
# Linux needs their development packages. Without them the sample builds silent and only prints audio commands.

option(OPENBONE_FETCH_SFML "Download and build SFML 2.6 when it is not installed" OFF)
option(OPENBONE_SFML_AUDIO "Play the document's sounds with sfml-audio when it can be built or found" ON)

set(OB_SFML_AUDIO OFF)
if(OPENBONE_SFML_AUDIO)
  find_package(SFML 2.6 COMPONENTS graphics window system audio QUIET)
  if(SFML_FOUND)
    set(OB_SFML_AUDIO ON)
  endif()
endif()
if(NOT SFML_FOUND)
  find_package(SFML 2.6 COMPONENTS graphics window system QUIET)
endif()
if(NOT SFML_FOUND)
  if(NOT OPENBONE_FETCH_SFML)
    message(FATAL_ERROR "SFML 2.6 not found: install it (e.g. libsfml-dev) or configure with -DOPENBONE_FETCH_SFML=ON")
  endif()
  if(OPENBONE_SFML_AUDIO)
    find_package(OpenAL QUIET)
    find_path(OB_VORBIS_INCLUDE vorbis/vorbisfile.h)
    find_path(OB_FLAC_INCLUDE FLAC/stream_decoder.h)
    if(OPENAL_FOUND AND OB_VORBIS_INCLUDE AND OB_FLAC_INCLUDE)
      set(OB_SFML_AUDIO ON)
    else()
      message(WARNING "OpenAL / Vorbis / FLAC development files not found: building SFML without audio (the sample is silent)")
    endif()
  endif()
  include(FetchContent)
  set(BUILD_SHARED_LIBS OFF)
  set(SFML_BUILD_AUDIO ${OB_SFML_AUDIO} CACHE BOOL "" FORCE)
  set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(sfml URL https://github.com/SFML/SFML/archive/refs/tags/2.6.2.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(sfml)
endif()

set(OPENBONE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(OPENBONE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../.. openbone EXCLUDE_FROM_ALL)

add_executable(openbone_sfml main.cpp)
target_include_directories(openbone_sfml PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../common)
target_link_libraries(openbone_sfml PRIVATE sfml-graphics sfml-window sfml-system openbone::openbone)
if(OB_SFML_AUDIO)
  target_link_libraries(openbone_sfml PRIVATE sfml-audio)
  target_compile_definitions(openbone_sfml PRIVATE OPENBONE_SFML_AUDIO=1)
else()
  target_compile_definitions(openbone_sfml PRIVATE OPENBONE_SFML_AUDIO=0)
endif()
message(STATUS "OpenBone SFML sample: sound ${OB_SFML_AUDIO}")
target_compile_features(openbone_sfml PRIVATE cxx_std_17)
